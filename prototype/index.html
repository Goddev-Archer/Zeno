use std::time::Instant;

fn benchmark_encryption() {
    let key = sframe::generate_key();
    let payload = vec![0u8; 960]; // 20ms Opus frame

    let start = Instant::now();
    for _ in 0..10_000 {

    }
    println!("10k encryptions: {:?}", start.elapsed());
}
